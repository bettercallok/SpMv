cmake_minimum_required(VERSION 3.18)
project(cuda_spmv_formats LANGUAGES CXX)
option(SPMV_CPU_ONLY "Build only the host library and CPU tests (no CUDA needed)" OFF)
option(SPMV_USE_LDG "Read x through the read-only cache (__ldg)" OFF)
set(CMAKE_CXX_STANDARD 17)
if(NOT CMAKE_BUILD_TYPE)
  set(CMAKE_BUILD_TYPE Release)
endif()

add_library(spmv_host STATIC src/mmio.cpp src/convert.cpp src/generate.cpp src/spmv_cpu.cpp)
target_include_directories(spmv_host PUBLIC include)
enable_testing()
add_executable(test_cpu tests/test_cpu.cpp)
target_link_libraries(test_cpu PRIVATE spmv_host)
target_compile_definitions(test_cpu PRIVATE TEST_DATA_DIR="${CMAKE_CURRENT_SOURCE_DIR}/tests/data")
add_test(NAME test_cpu COMMAND test_cpu)
if(SPMV_CPU_ONLY)
  return()
endif()

# Target this machine's GPU (CMake >= 3.24); override with e.g. -DCMAKE_CUDA_ARCHITECTURES=75 (T4).
if(NOT DEFINED CMAKE_CUDA_ARCHITECTURES)
  set(CMAKE_CUDA_ARCHITECTURES native)
endif()
enable_language(CUDA)
set(CMAKE_CUDA_STANDARD 17)
find_package(CUDAToolkit REQUIRED)
add_executable(spmv_bench src/spmv_csr.cu src/spmv_ell.cu src/spmv_coo.cu src/spmv_hyb.cu src/bench.cu)
target_link_libraries(spmv_bench PRIVATE spmv_host CUDA::cudart CUDA::cusparse)
target_compile_options(spmv_bench PRIVATE $<$<COMPILE_LANGUAGE:CUDA>:-lineinfo>)  # source view in Nsight
if(SPMV_USE_LDG)
  target_compile_definitions(spmv_bench PRIVATE SPMV_USE_LDG)
endif()
